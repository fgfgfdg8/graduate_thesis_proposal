\begin{algorithm}[htbp]
\caption{证据条件的调用调解与恢复（\textsc{MediateAndRecover}）}
\label{alg:trace_cut}
\KwIn{工具调用提案 $\text{proposal}$，证据图 $G_t$，任务授权证据 $K_t$}
\KwOut{真实执行结果，或重规划/恢复后的安全继续结果}
$a \leftarrow \text{NormalizeAndValidateInterface}(\text{proposal})$\;
$h \leftarrow \text{EvaluateAuthorization}(K_t,a)$\;
$\mathcal{P}_t \leftarrow \text{EvaluateProvenance}(G_t,a)$ \tcp{按式\eqref{eq:trace_path}生成路径证据包}
$c_t \leftarrow \text{InvalidTool}(a)\lor h.\text{blocked}\lor(\mathcal{P}_t\neq\varnothing)$\;
\uIf{$c_t=0$}{
  $a'\leftarrow a$\;
}
\Else{
  $B_t\leftarrow\text{BoundedEvidence}(q,a,\text{Schema}(a),h,\mathcal{P}_t,\text{RecentDecisions})$\;
  $d\leftarrow\text{SemanticMediator}(B_t)$ \tcp{仅输出\texttt{ALLOW}/\texttt{REWRITE}/\texttt{REPLAN}}
  \uIf{$d=\texttt{REWRITE}$}{
    $a'\leftarrow\text{SchemaValidate}(d.\text{rewrittenArgs})$\;
  }
  \uElseIf{$d=\texttt{ALLOW}$}{$a'\leftarrow a$\;}
  \Else{$a'\leftarrow\bot$\;}
  \If{$d=\texttt{REWRITE}$且$\text{EvidenceClosure}(K_t,G_t,a')=0$}{$a'\leftarrow\bot$\; $d\leftarrow\texttt{REPLAN}$\;}
  \If{$d=\texttt{ALLOW}$且$\neg\text{SemanticAllowEligible}(a,B_t,d)$}{$a'\leftarrow\bot$\; $d\leftarrow\texttt{REPLAN}$\;}
}
\uIf{$a'\neq\bot$}{
  $cp\leftarrow\text{CaptureCheckpoint}(\text{Env},G_t,K_t,\text{Lineage})$\;
  $y\leftarrow\text{InvokeRealFunctionOnce}(a')$\;
  $G_{t+1}\leftarrow\text{ObserveCallAndRawResult}(G_t,a',y)$\;
  记录动作摘要、检查点、调解决策、时延与执行回执\;
  \Return{格式化工具结果}\;
}
\Else{
  \If{$\mathcal{P}_t\neq\varnothing$}{
    $r\leftarrow\text{RestoreEarliestReversibleCheckpoint}(\mathcal{P}_t)$\;
    \If{路径已经过外部汇聚}{记录\texttt{unrecoverable\_external\_effect}\;}
    将$\mathcal{P}_t$中的来源写入隔离表$Q_t$\;
  }
  返回恢复提示；任何新调用重新执行本算法\;
}
\end{algorithm}
