\begin{algorithm}[htbp]
\caption{TGG统一在线安全检查算法（\textsc{TGGCheck}）}
\label{alg:contract_gate}
\KwIn{当前物化动作$M_t$，运行时事件，图$G_{t-1}$，任务契约$C$}
\KwOut{更新图$G_t$，规则见证，决策$Decision_t$}
\SetKwFunction{FCheck}{TGGCheck}
\SetKwProg{Fn}{Function}{:}{}
\Fn{\FCheck{$M_t$, events, $G_{t-1}$, $C$}}{
  \tcp{阶段一：增量图构建}
  将\texttt{AgentStep}、\texttt{ToolCall}、\texttt{ArgumentSlot}等节点写入$G_t$\;
  添加\texttt{OBSERVED+HARD}结构边\;
  \tcp{阶段二：确定性边推导}
  运行\textsc{EXACT-VALUE}、\textsc{STRUCTURED-ALIAS}、\textsc{OBJECT-VERSION}、\textsc{STATE-DIFF}推导器\;
  对未解析参数与新分支生成\texttt{SOFT/UNKNOWN}候选边\;
  \tcp{阶段三：契约规则检查}
  查询五类图检查规则（见\autoref{tab:contract_actions}）\;
  \uIf{$\exists$ 硬证据违规路径}{
    $Decision_t \leftarrow \textsc{Block}$\;
    签发阻断回执\;
  }
  \uElseIf{不存在硬违规但存在高影响\texttt{SOFT/UNKNOWN}关系}{
    \tcp{阶段四：选择性LLM补边}
    批量调用LLM对未解析关系输出结构化\texttt{SOFT}边\;
    确定性验证器独立重算，通过则生成新\texttt{HARD}边\;
    重新查询规则\;
    \uIf{补边后仍存在未解析高影响关系}{
      \uIf{来源距离$\le 2$步且存在干净检查点}{
        $Decision_t \leftarrow \textsc{Rewrite}$\;
        隔离净化来源$\rightarrow$局部重执行\;
      }
      \Else{
        $Decision_t \leftarrow \textsc{Replan}$\;
        语义调解器选择\textsc{Rollback}/\textsc{Withdraw}/\textsc{Abort}\;
      }
    }
    \Else{按补边后规则结果重新判定\;}
  }
  \Else{
    \If{$Authorized_C(M_t)=1$}{
      $Decision_t \leftarrow \textsc{Allow}$\;
      $result \leftarrow \text{InvokeRealFunctionOnce}(M_t)$\;
      记录\texttt{ToolResult}、\texttt{Effect}回执与状态差分\;
      \Return{$result$及执行回执}\;
    }
  }
  恢复或阻断后的任何新动作重新调用\FCheck{}\;
}
\end{algorithm}
